     1|"""

     8|  效果: 认识的模式 → 低预测误差 → 不触发好奇心
     9|        不认识的模式 → 高预测误差 → 触发演化

    14|import math
    15|
    16|DIM = 128
    17|
    18|
    19|class PatternMemory:
    20|    """存储和匹配已知向量模式。这是识别的基础。"""
    21|
    22|    def __init__(self, similarity_threshold=0.15):
    23|        self.patterns = []  # [{id, vector, delta, hits, first_seen, last_seen, label}]
    24|        self.threshold = similarity_threshold
    25|        self.next_id = 0
    26|        self.total_matches = 0
    27|        self.total_novels = 0
    28|
    29|    def match(self, vector):
    30|        """
    31|        查找最接近的已知模式。
    32|        返回 (matched_pattern_dict, similarity) 或 (None, 0)
    33|        similarity: 0=完全不同, 1=完全匹配
    34|        """
    35|        if len(self.patterns) == 0:
    36|            self.total_novels += 1
    37|            return None, 0.0
    38|
    39|        best = None
    40|        best_sim = -1.0
    41|
    42|        for p in self.patterns:
    43|            # 余弦相似度
    44|            dot = np.dot(vector, p["vector"])
    45|            norm_a = np.linalg.norm(vector)
    46|            norm_b = np.linalg.norm(p["vector"])
    47|            if norm_a < 1e-8 or norm_b < 1e-8:
    48|                sim = 0
    49|            else:
    50|                sim = dot / (norm_a * norm_b)
    51|
    52|            if sim > best_sim:
    53|                best_sim = sim
    54|                best = p
    55|
    56|        if best_sim >= 1.0 - self.threshold:
    57|            self.total_matches += 1
    58|            return best, best_sim
    59|        else:
    60|            self.total_novels += 1
    61|            return None, best_sim
    62|
    63|    def learn(self, vector, delta, label=""):
    64|        """
    65|        记录新的模式或更新已有模式。
    66|        返回 (pattern_id, is_new)
    67|        """
    68|        matched, sim = self.match(vector)
    69|
    70|        if matched is not None:
    71|            # 更新已有模式
    72|            matched["hits"] += 1
    73|            matched["last_seen"] = time.time()
    74|            # 逐渐调整预期Δ
    75|            alpha = 1.0 / matched["hits"]
    76|            matched["delta"] = (1 - alpha) * matched["delta"] + alpha * delta
    77|            return matched["id"], False
    78|        else:
    79|            # 新模式
    80|            pid = self.next_id
    81|            self.next_id += 1
    82|            self.patterns.append({
    83|                "id": pid,
    84|                "vector": vector.copy(),
    85|                "delta": delta.copy(),
    86|                "hits": 1,
    87|                "first_seen": time.time(),
    88|                "last_seen": time.time(),
    89|                "label": label
    90|            })
    91|            return pid, True

    94|        return len(self.patterns)
